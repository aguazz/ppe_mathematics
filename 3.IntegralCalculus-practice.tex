\begin{frame}{Definite integrals}

    \begin{enumerate}[leftmargin = 0.5cm, label = \arabic*., rightmargin = 0.2cm]
        
        \item Find the function $f(x)$ such that
        \begin{enumerate}[leftmargin = 0.5cm, label = (\roman*), rightmargin = 0.2cm]
        
            \item $f'(x) = ax^2 + bx$,

            \item $f'(1) = 6$,

            \item $f''(1) = 18$,

            \item $\int_0^2 f(x)\,\rmd x = 18$.
            
        \end{enumerate}

        \item If $\int_0^1 (f(x) - 2g(x))\,\rmd x = 6$ and $\int_0^1 (2f(x) + 2g(x))\,\rmd x = 9$, find $\int_0^1 (f(x) - g(x))\,\rmd x$.

        \item If $\int_a^b f(x)\,\rmd x = 8$ and $\int_a^c f(x)\,\rmd x = 5$, find the value of $\int_c^b f(x)\,\rmd x$.
        
    \end{enumerate}
    
\end{frame}

\begin{frame}{Definite integrals — Solution}
\small
\begin{enumerate}[leftmargin=0.5cm,label=\arabic*.,rightmargin=0.2cm]
      \item 
      From $f'(x) = ax^2 + bx$, $f'(1)=6$ and $f''(1)=18$:
      \[
      a+b=6,\quad 2a+b=18 \;\Rightarrow\; a=12,\; b=-6.
      \]
      Thus $f'(x)=12x^2-6x$ and
      \[
      f(x)=\int (12x^2-6x)\,\mathrm{d}x=4x^3-3x^2+C.
      \]
      Using $\int_0^2 f(x)\,\mathrm{d}x=18$:
      \[
      \big[x^4-x^3+Cx\big]_0^2=16-8+2C=8+2C=18 \Rightarrow C=5.
      \]
      \[
      \boxed{\,f(x)=4x^3-3x^2+5\,}
      \]
    
      \item Let $A=\int_0^1 f(x)\,\mathrm{d}x$, $B=\int_0^1 g(x)\,\mathrm{d}x$.
      \[
      A-2B=6,\quad 2A+2B=9.
      \]
      Hence, $B=-\tfrac12$, $A=5$, meaning that
      \[
      \boxed{\,\int_0^1 (f-g)\,\mathrm{d}x=A-B=\tfrac{11}{2}\,}
      \]
    
      \item \[
      \boxed{\,\int_c^b f(x)\,\mathrm{d}x=\int_a^b f(x)\,\mathrm{d}x - \int_a^c f(x)\,\mathrm{d}x =8-5=3\,}
      \]
    \end{enumerate}
\end{frame}


\begin{frame}{Areas delimited by positive graphs}

    Compute the area bounded by the graph of the function over the indicated interval. Sketch the graph

    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]

        \item $f(x) = 3x^2$, in $[0, 2]$

        \item $f(x) = e^{x}$, in $[-1, 1]$

        \item $f(x) = x^6$, in $[0, 1]$

        \item $f(x) = 1/x^2$, in $[1, 10]$
        
    \end{enumerate}
    
\end{frame}

\begin{frame}{Areas delimited by positive graphs  —  Solution}
\small
For each positive $f$ on $[a,b]$, the area is $\displaystyle \int_a^b f(x)\,\mathrm{d}x$.
\begin{enumerate}[leftmargin=0.5cm,label=(\alph*),rightmargin=0.2cm]
  \item $f(x)=3x^2$ on $[0,2]$:
  \[
  \int_0^2 3x^2\,\mathrm{d}x=\left[x^3\right]_0^2=\boxed{8}.
  \]

  \item $f(x)=e^{x}$ on $[-1,1]$:
  \[
  \int_{-1}^{1} e^x\,\mathrm{d}x=\left[e^x\right]_{-1}^{1}=\boxed{e-\tfrac{1}{e}}.
  \]

  \item $f(x)=x^6$ on $[0,1]$:
  \[
  \int_0^1 x^6\,\mathrm{d}x=\left[\tfrac{x^7}{7}\right]_0^1=\boxed{\tfrac{1}{7}}.
  \]

  \item $f(x)=\dfrac{1}{x^2}$ on $[1,10]$:
  \[
  \int_1^{10} x^{-2}\,\mathrm{d}x=\left[-\tfrac{1}{x}\right]_1^{10}=1-\tfrac{1}{10}=\boxed{\tfrac{9}{10}}.
  \]
\end{enumerate}
\end{frame}

\begin{frame}{Areas delimited by graphs}

    Compute the area bounded by the graph of the function over the indicated interval. Be mindful of the sign of each function.

    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]

        \item $f(x) = x$, in $[-2, 2]$

        \item $f(x) = (e^{x} + e^{-x})/2$, in $[-1, 1]$

        \item $f(x) = \cos(x)$, in $[0, 2\pi]$
        
    \end{enumerate}
    
\end{frame}

\begin{frame}{Areas delimited by graphs — Solution}
\small
For signed functions, the area on $[a,b]$ is $\displaystyle \int_a^b |f(x)|\,\mathrm{d}x$.
\begin{enumerate}[leftmargin=0.5cm,label=(\alph*),rightmargin=0.2cm]
  \item $f(x)=x$ on $[-2,2]$ (negative on $[-2,0]$, positive on $[0,2]$):
  \[
  \int_{-2}^2 |x|\,\mathrm{d}x=2\int_0^2 x\,\mathrm{d}x=2\cdot\Big[\tfrac{x^2}{2}\Big]_0^2=\boxed{4}.
  \]

  \item $f(x)=\dfrac{e^x+e^{-x}}{2}$ on $[-1,1]$:
    \[
    \int_{-1}^{1}\frac{e^x+e^{-x}}{2}\,dx
    =\frac12\Big[e^x - e^{-x}\Big]_{-1}^{1}
    =\frac12\Big((e - e^{-1})-(e^{-1}-e)\Big)
    =\boxed{\,e-\tfrac{1}{e}\,}.
    \]

  \item $f(x)=\cos x$ on $[0,2\pi]$ (changes sign at $\tfrac{\pi}{2}$ and $\tfrac{3\pi}{2}$):
  \[
  \int_0^{2\pi}\!|\cos x|\,\mathrm{d}x
  =\int_0^{\pi/2}\!\cos x\,\mathrm{d}x+\int_{\pi/2}^{3\pi/2}\!(-\cos x)\,\mathrm{d}x+\int_{3\pi/2}^{2\pi}\!\cos x\,\mathrm{d}x
  =1+2+1=\boxed{4}.
  \]
\end{enumerate}
\end{frame}

\begin{frame}{Lorenz curve and Gini index}

    Answer the following questions
    \begin{enumerate}[leftmargin = 0.5cm, label = \arabic*., rightmargin = 0.2cm]
        \item Reason why a Lorenz curve $L(x)$ must satisfy the following conditions
        \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]
            \item $L(0) = 0$, $L(1) = 1$
            \item $L(x)$ is non-decreasing
            \item $L(x) \leq x$
        \end{enumerate}
        \item Why does the case $L(x) = x$ is called perfect equality?
        \item Show that the following functions are valid Lorenz curves
        \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]
            \item $L(x) = x^p$, $p \geq 1$
            \item $L(x) = 5x^2/6 + x/6$
        \end{enumerate}
        \item Compute the Gini index associated to each one of the previous Lorenz curves
        \item According to the Gini index, determine which case is more ``unequal'' and quantify that inequality gap:
        \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]
            \item $L_1(x) = x^2$ versus $L_2(x) = x^5$
            \item $L_1(x) = 5x^2/6 + x/6$ vs $L_2(x) = x^2$
        \end{enumerate}
    \end{enumerate}
\end{frame}

\begin{frame}{Lorenz curve and Gini index — Solution (1/2)}
\small
\textbf{1. Properties of a Lorenz curve $L(x)$}
\begin{enumerate}[leftmargin=0.5cm,label=(\alph*),rightmargin=0.2cm]
  \item $L(0)=0$, $L(1)=1$: with $0\%$ of the population, wealth share is $0\%$; with all the population, wealth share is $100\%$.
  \item Non-decreasing: adding people cannot reduce cumulative wealth share $\Rightarrow L'(x)\ge 0$.
  \item $L(x)\le x$: the bottom (poorest) $x$ fraction cannot own more than an $x$ share of wealth.
\end{enumerate}

\textbf{2. Why $L(x)=x$ is ``perfect equality''?} Each bottom $x$ fraction owns exactly $x$ share of wealth $\Rightarrow$ everyone has the same wealth.

\textbf{3. Validity checks}
\begin{enumerate}[leftmargin=0.5cm,label=(\alph*),rightmargin=0.2cm]
  \item $L(x)=x^p$, $p\ge1$: $L(0)=0$, $L(1)=1$; $L'(x)=p x^{p-1}\ge0$; and $x^p\le x$ on $[0,1]$.
  \item $L(x)=\frac{5}{6}x^2+\frac{1}{6}x$: $L(0)=0$, $L(1)=1$; $L'(x)=\frac{5}{3}x+\frac{1}{6}\ge0$; and
  \[
  x^2 \leq x \text{ for } x\in[0, 1] \Longrightarrow \tfrac{5}{6}x^2+\tfrac{1}{6}x\le x \text{ for } x\in[0, 1] \Longrightarrow L(x)\le x \text{ on } [0,1].
  \]
\end{enumerate}
\end{frame}

\begin{frame}{Lorenz curve and Gini index — Solution (2/2)}
\small
\textbf{4. Gini index:} \(\displaystyle G=1-2\int_0^1 L(x)\,dx\).

\medskip
\begin{tabular}{@{}l l l@{}}
(a) $L(x)=x^p$ & $\displaystyle \int_0^1 x^p dx=\frac{1}{p+1}$ & 
\(\displaystyle \boxed{G=\frac{p-1}{p+1}}\).\\[6pt]
(b) $L(x)=\frac{5}{6}x^2+\frac{1}{6}x$ & 
$\displaystyle \int_0^1 L=\frac{5}{18}+\frac{1}{12}=\frac{13}{36}$ &
\(\displaystyle \boxed{G=1-2\cdot\frac{13}{36}=\frac{5}{18}}\).
\end{tabular}

\medskip
\textbf{5. Inequality comparisons (via $G$)}
\begin{enumerate}[leftmargin=0.5cm,label=(\alph*),rightmargin=0.2cm]
  \item $x^2$: \(G=\frac{1}{3}\), \quad $x^5$: \(G=\frac{4}{6}=\frac{2}{3}\).
  
  \(\Rightarrow\) \(\boxed{x^5 \text{ is more unequal by } \frac{2}{3}-\frac{1}{3}=\frac{1}{3}}\).
  
  \item $\frac{5}{6}x^2+\frac{1}{6}x$: \(G=\frac{5}{18}\), \quad $x^2$: \(G=\frac{1}{3}\).
  
  \(\Rightarrow\) \(\boxed{x^2 \text{ is more unequal by } \frac{1}{3}-\frac{5}{18}=\frac{1}{18}}\).
\end{enumerate}
\end{frame}

\begin{frame}{Finding the Lorenz curve.}
    Suppose the distribution of income for of civil engineers in a specific region is given by the following Lorenz curve:  
    \[
    L(x) = \frac{3}{4}x^p + ax.
    \]  
    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]
            \item Find\footnote{
            \mbox{Recall that $L(x)$ represents the proportion of overall wealth held by the poorest $x\%$ of the population (e.g., $L(0.5) = 0.2$} 
            indicates that  the poorest $50\%$ of the population holds $20\%$ of the wealth).
            } the value of $a$.
            \item Given the definition of the Gini index as $G = A/(A+B)$, where:
            \begin{enumerate}[leftmargin = 0.5cm, label = $\bullet$, rightmargin = 0.2cm]
                \item $A$: area between $f(x) = x$ and the Lorenz curve $L(x)$, and 
                \item $B$: area under the Lorenz curve. 
            \end{enumerate}
            Prove that $G = 2 \int_0^1 (x - L(x)) \, \mathrm{d}x$.
            \item Provided that the Gini index is $3/8$, find the value of $p$.
    \end{enumerate}
\end{frame}

\begin{frame}{Finding the Lorenz curve — Solutions}
    \small
    Given $L(x)=\tfrac{3}{4}x^{p}+ax$.
    
    \textbf{(a) Determine $a$.}
    Lorenz curves satisfy $L(0)=0$ and $L(1)=1$.
    \[
    L(1)=\tfrac{3}{4}\cdot 1^{p}+a\cdot 1=1 \;\Rightarrow\; a=\boxed{\tfrac{1}{4}}.
    \]
    Hence $L(x)=\tfrac{3}{4}x^{p}+\tfrac{1}{4}x$.
    
    \medskip
    \textbf{(b) Show } $\displaystyle G=2\int_{0}^{1}\big(x-L(x)\big)\,dx$.
    Let $A=\int_{0}^{1}\big(x-L(x)\big)\,dx$ (area between $y=x$ and $L$), $B=\int_{0}^{1}L(x)\,dx$.
    Then $A+B=\int_{0}^{1}x\,dx=\tfrac{1}{2}$, so with $G=\dfrac{A}{A+B}$,
    \[
    G=\frac{A}{1/2}=2A=2\int_{0}^{1}\big(x-L(x)\big)\,dx.
    \]
    
    \medskip
    \textbf{(c) Given } $G=\tfrac{3}{8}$, find $p$.
    \[
    \int_{0}^{1} L(x)\,dx=\tfrac{3}{4}\cdot\frac{1}{p+1}+\tfrac{1}{4}\cdot\frac{1}{2}
    =\frac{3}{4(p+1)}+\frac{1}{8}.
    \]
    Using $G=1-2\int_{0}^{1}L(x)\,dx$,
    \[
    \frac{3}{8}=1-2\!\left(\frac{3}{4(p+1)}+\frac{1}{8}\right)
    =\frac{3}{4}-\frac{3}{2(p+1)}
    \;\Rightarrow\;
    \frac{3}{2(p+1)}=\frac{3}{8}\;\Rightarrow\; p+1=4.
    \]
    \[
    \boxed{p=3}.
    \]
\end{frame}



\begin{frame}{Consumer surplus. Linear case}
    Let the demand and supply sets be
    \[
    D= \{(q,p)\in\R_+^2: q=100-2p\}
    \qquad
    S= \{(q,p)\in\R_+^2: q=3p-10\}.
    \]
    Find the consumer surplus.
\end{frame}

\begin{frame}{Consumer surplus. Linear case — Solution}
    \small
    Demand: $p_D(q)=50-\tfrac{q}{2}$,\quad Supply: $p_S(q)=\tfrac{q+10}{3}$.
    
    \textbf{Equilibrium.} Solve $p_D=p_S$:
    \[
    50-\frac{q}{2}=\frac{q+10}{3}\;\Rightarrow\; 300-3q=2q+20 \;\Rightarrow\; q^*=56,\quad
    p^*=22.
    \]
    
    \textbf{Consumer surplus.} Area under demand above $p^*$ on $[0,q^*]$:
    \[
    \text{CS}=\int_{0}^{56}\!\big(50-\tfrac{q}{2}-22\big)\,dq
    =\int_{0}^{56}\!\big(28-\tfrac{q}{2}\big)\,dq
    =\Big[28q-\tfrac{q^{2}}{4}\Big]_{0}^{56}
    =784.
    \]
    (Equivalently, triangle area $\tfrac{1}{2}\cdot 56\cdot(50-22)=\tfrac{1}{2}\cdot56\cdot28=784$.)
    
    \[
    \boxed{\text{Consumer surplus }=784}
    \]
\end{frame}

\begin{frame}{Consumer surplus. Parabolic case}
    Let the demand and supply sets be
    $$
    D= \{(q,p)\ |\ p + q^2 +2q = 15\},\qquad S = \{(q,p)\ |\ 5p- q^2 -2q = 27\}.
    $$
    Find the consumer surplus.
\end{frame}

\begin{frame}{Consumer surplus. Parabolic case — Solution}
    \small
    Supply: $5p-q^2-2q=27 \;\Rightarrow\; p_S(q)=\dfrac{q^2+2q+27}{5}$, \\
    Demand: $p+q^2+2q=15 \;\Rightarrow\; p_D(q)=15-q^2-2q$.
    
    \textbf{Equilibrium.} Set $p_D=p_S$:
    \[
    15-q^2-2q=\frac{q^2+2q+27}{5}
    \;\Rightarrow\; 6q^2+12q-48=0
    \;\Rightarrow\; (q+4)(q-2)=0.
    \]
    Feasible $q^*=2$, and $p^*=p_D(2)=15-4-4=7$.
    
    \textbf{Consumer surplus.}
    \[
    \text{CS}=\int_{0}^{q^*}\!\big(p_D(q)-p^*\big)\,dq
    =\int_{0}^{2}\!\big(15-q^2-2q-7\big)\,dq
    =\int_{0}^{2}\!(8-q^2-2q)\,dq
    \]
    \[
    =\Big[8q-\tfrac{q^3}{3}-q^2\Big]_{0}^{2}
    =16-\tfrac{8}{3}-4
    =\boxed{\tfrac{28}{3}}.
    \]
\end{frame}

\begin{frame}{Improper integrals}

    Compute those integrals that are convergent and indicate those that diverge.

    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]
        
        \item $\int_1^\infty \frac{1}{x^3}\,\rmd x$.

        \item $\int_1^\infty \frac{1}{\sqrt{x}}\,\rmd x$.

        \item $\int_{-\infty}^0 e^x\,\rmd x$.

        \item $\int_0^a \frac{x}{\sqrt{a^2 - x^2}}\,\rmd x$.

        \item $\int_{-2}^3 \left(\frac{1}{\sqrt{x + 2}} + \frac{1}{\sqrt{3 - x}}\right)\, \rmd x$.
   
    \end{enumerate}
    
\end{frame}

\begin{frame}{Improper integrals — Solutions}
    \small
    \begin{enumerate}[leftmargin=0.5cm,label=(\alph*),rightmargin=0.2cm]
      \item $\displaystyle \int_1^\infty \frac{1}{x^3}\,dx
      =\Big[-\frac{1}{2x^2}\Big]_{1}^{\infty}=0-\!\Big(-\frac12\Big)=\boxed{\tfrac12}$ (convergent).
    
      \item $\displaystyle \int_1^\infty \frac{1}{\sqrt{x}}\,dx
      =\Big[2\sqrt{x}\Big]_{1}^{\infty}=+\infty$ \;\;$\Rightarrow$ \;\;\boxed{\text{diverges}}.
    
      \item $\displaystyle \int_{-\infty}^{0} e^x\,dx
      =\Big[e^x\Big]_{-\infty}^{0}=1-0=\boxed{1}$ (convergent).
    
      \item $\displaystyle \int_0^a \frac{x}{\sqrt{a^2-x^2}}\,dx$.
      Let $u=a^2-x^2$, $du=-2x\,dx$:
      \[
      \frac12\!\int_{u= a^2}^{0}\!\!\frac{-du}{\sqrt{u}}
      =\frac12\!\int_{0}^{a^2}\!u^{-1/2}\,du
      =\Big[\sqrt{u}\Big]_{0}^{a^2}=\boxed{a}\quad(\text{convergent}).
      \]
    
      \item $\displaystyle \int_{-2}^{3}\!\!\left(\frac{1}{\sqrt{x+2}}+\frac{1}{\sqrt{3-x}}\right)\!dx$:
      \[
      \int_{-2}^{3}\!\frac{1}{\sqrt{x+2}}\,dx=2\sqrt{x+2}\Big|_{-2}^{3}=2\sqrt{5},
      \]
      \[
      \int_{-2}^{3}\!\frac{1}{\sqrt{3-x}}\,dx=-2\sqrt{3-x}\Big|_{-2}^{3}=2\sqrt{5}.
      \]
      Sum $=2\sqrt{5}+2\sqrt{5}=\boxed{4\sqrt{5}}$ (convergent).
    \end{enumerate}
\end{frame}

\begin{frame}{Improper integrals}

    \begin{enumerate}[leftmargin = 0.5cm, label = \arabic*., rightmargin = 0.2cm]
        \item For what values of $a\in \R$ does the integral $\int_1^\infty 1/x^a\,\rmd x$ converges/diverges?

        \item For what values of $a\in \R$ does the integral $\int_0^1 1/x^a\,\rmd x$ converges/diverges?
    \end{enumerate}
    
\end{frame}

\begin{frame}{Convergent/divergent improper integrals}
    For numbers $a, p \in\mathbb{R}$, define the function
    \begin{align*}
        f_{a,p}(x) =
        \begin{cases}
        ax & x\in [0, 1] \\
        x^{-p} & x\in(1,\infty)
        \end{cases}
    \end{align*}

    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]
        
        \item What is the value of $a$ for which $\int_0^\infty f_{a, 2}(x)\,\mathrm{d}x = 1$?

        \item What is the value of $\int_0^\infty f_{a, 1}(x)\,\mathrm{d}x$?
        
    \end{enumerate}
\end{frame}

\begin{frame}{Convergent/divergent improper integrals — Solutions}
    \small
    Given
    \[
    f_{a,p}(x)=
    \begin{cases}
    ax, & x\in[0,1],\\
    x^{-p}, & x\in(1,\infty).
    \end{cases}
    \]
    
    \begin{enumerate}[leftmargin=0.5cm,label=(\alph*),rightmargin=0.2cm]
    \item For $p=2$:
    \[
    \int_{0}^{\infty} f_{a,2}(x)\,dx=\int_{0}^{1} ax\,dx+\int_{1}^{\infty} x^{-2}\,dx
    =\frac{a}{2}+1.
    \]
    Set equal to $1$:
    \[
    \frac{a}{2}+1=1 \;\Rightarrow\; \boxed{a=0}.
    \]
    
    \item For $p=1$:
    \[
    \int_{0}^{\infty} f_{a,1}(x)\,dx=\int_{0}^{1} ax\,dx+\int_{1}^{\infty} x^{-1}\,dx
    =\frac{a}{2}+\lrb{\ln(x)}_1^\infty = \infty.
    \]
    Hence, for all $a$:
    \[
    \boxed{\int_{0}^{\infty} f_{a,1}(x)\,dx=\infty\ \text{(diverges)}}.
    \]
    \end{enumerate}
\end{frame}
    

\begin{frame}{Comparison test for convergence of integrals}

    Prove that the following integrals converge by using the comparison test:

    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]
        
        \item $\int_1^\infty \frac{1}{1 + x^2}\,\rmd x$. Compare with $\int_1^\infty \frac{1}{x^2}\,\rmd x$

        \item $\int_1^\infty e^{-x^2}\,\rmd x$. Compare with $\int_1^\infty e^-x\,\rmd x$

    \end{enumerate}
    
\end{frame}

\begin{frame}{Comparison test for convergence — Solution}
    \small
    \begin{enumerate}[leftmargin=0.5cm,label=(\alph*),rightmargin=0.2cm]
      \item 
      For $x\ge 1$, $\dfrac{1}{1+x^{2}}\le \dfrac{1}{x^{2}}$ and $\displaystyle \int_{1}^{\infty}\frac{1}{x^{2}}\,dx$ converges, hence, by comparison, the integral converges:
        $$
        0 \leq \int_{1}^{\infty}\frac{1}{1+x^{2}}\,dx \leq\int_{1}^{\infty}\frac{1}{x^{2}}\,dx = \lrb{-\frac{1}{x}}_1^\infty = 1.
        $$
      \item 
      For $x\ge 1$, $x^{2}\ge x \Rightarrow e^{-x^{2}}\le e^{-x}$, so
      \[
      0 \leq \int_{1}^{\infty}e^{-x^{2}}\,dx \leq \int_{1}^{\infty}e^{-x}\,dx = \lrb{-e^{-x}}_1^\infty = e^{-1}.
      \]
    \end{enumerate}
\end{frame}
