\begin{frame}{Determinate forms of limits and the rule of limits}

    Find the limits:

    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]
        
        \item $\lim_{x\rightarrow \infty}\frac{1}{x^2 + 3x}$

        \item $\lim_{x\rightarrow 0}\frac{x^2 - 1}{x^2}$

        \item $\lim_{x\rightarrow 2}\frac{\sin(x - 2) + \cos(x - 2)}{x - 2}$

        \item $\lim_{x\rightarrow \infty} \ln(x^x)$

        \item $\lim_{x\rightarrow 0} xe^x$

    \end{enumerate}
    
\end{frame}

\begin{frame}{Determinate forms of limits — Solutions}

\begin{enumerate}[leftmargin=0.5cm, label=(\alph*), rightmargin=0.2cm]

\item $\displaystyle \lim_{x\to\infty}\frac{1}{x^2+3x}=0$
\quad (denominator growths indefinitely).

\item $\displaystyle \lim_{x\to 0}\frac{x^2-1}{x^2}=-\infty$
\quad (numerator $\to -1$, denominator $\to 0^+$).

\item $\displaystyle \lim_{x\to 2}\frac{\sin(x-2)+\cos(x-2)}{x-2}$ \\
(numerator $\to \sin(0)+\cos(0) = 0 + 1 = 1$, denominator $\to \pm\infty$) \\
% set $h=x-2\to0$,
% \[
% \frac{\sin h+\cos h}{h}=\frac{\sin h}{h}+\frac{\cos h}{h}
% \to 1+\text{(diverges)}.
% \]
No limit: $+\infty$ from the right and $-\infty$ from the left.

\item $\displaystyle \lim_{x\to\infty}\ln(x^x)
=\lim_{x\to\infty}x\ln x=+\infty$.

\item $\displaystyle \lim_{x\to 0} x e^x
= \big(\lim_{x\to0}x\big)\big(\lim_{x\to0}e^x\big)=0\cdot 1=0$.

\end{enumerate}

\end{frame}

\begin{frame}{Limits at $\pm\infty$ of ratios of polynomials}

Compute the following limits:

\begin{enumerate}[leftmargin=0.5cm, label=(\alph*), rightmargin=0.2cm]
\item $\displaystyle \lim_{x\to\infty}\frac{3x^2+3x-2}{9x-5}$
\item $\displaystyle \lim_{x\to-\infty}\frac{2x^3-x}{x^2+4}$
\item $\displaystyle \lim_{x\to-\infty}\frac{-7x^3+5}{2x^2-1}$
\item $\displaystyle \lim_{x\to\infty}\frac{4x-3}{5x^2+1}$
\item $\displaystyle \lim_{x\to-\infty}\frac{5x^4-x}{-10x^4+7}$
\end{enumerate}

\end{frame}

%==================== SOLUTIONS ====================
\begin{frame}{Limits at $\pm\infty$ — Solutions (leading-term rule)}

\small
% \textbf{Rule of thumb:}
% If $\deg N<\deg D\Rightarrow 0$;\;
% $\deg N=\deg D\Rightarrow$ ratio of leading coeffs;\;
% $\deg N>\deg D\Rightarrow \pm\infty$ with sign from leading terms.

\begin{enumerate}[leftmargin=0.5cm, label=(\alph*), rightmargin=0.2cm]

\item Divide by $x^2$:
\[
\frac{3x^2+3x-2}{9x-5}
=\frac{3+\frac{3}{x}-\frac{2}{x^2}}{\frac{9}{x}-\frac{5}{x^2}}
\;\xrightarrow[x\to\infty]{}\; \frac{3}{0^+}= \boxed{+\infty}.
\]

\item Divide by $x^3$:
\[
\frac{2x^3-x}{x^2+4}
=\frac{2-\frac{1}{x^2}}{\frac{1}{x}+\frac{4}{x^3}}
\;\xrightarrow[x\to-\infty]{}\; \frac{2}{0^-}= \boxed{-\infty}.
\]

\item Divide by $x^3$:
\[
\frac{-7x^3+5}{2x^2-1}
=\frac{-7+\frac{5}{x^3}}{\frac{2}{x}-\frac{1}{x^3}}
\;\xrightarrow[x\to-\infty]{}\; \frac{-7}{0^-}= \boxed{+\infty}.
\]

\item Divide by $x^2$:
\[
\frac{4x-3}{5x^2+1}
=\frac{\frac{4}{x}-\frac{3}{x^2}}{5+\frac{1}{x^2}}
\;\xrightarrow[x\to\infty]{}\; \frac{0}{5}= \boxed{0}.
\]

\item Divide by $x^4$:
\[
\frac{5x^4-x}{-10x^4+7}
=\frac{5-\frac{1}{x^3}}{-10+\frac{7}{x^4}}
\;\xrightarrow[x\to-\infty]{}\; \frac{5}{-10}= \boxed{-\tfrac12}.
\]

\end{enumerate}

\end{frame}

\begin{frame}{Indeterminate forms of limits: the factorization trick}

    Compute the limits:

    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]

        \item $\lim_{x\rightarrow 0} \frac{x^3 - 5x^2 + 3x}{2x}$

        \item $\lim_{x\to 1}\frac{x^2+7x-8}{x-1}$

    \end{enumerate}

\end{frame}

\begin{frame}{Indeterminate forms of limits: the factorization trick — Solution}

% \small
\begin{enumerate}[leftmargin=0.5cm, label=(\alph*), rightmargin=0.2cm]

\item $\displaystyle \lim_{x\to 0}\frac{x^3-5x^2+3x}{2x}$ \\
Factor out $x$:
\[
\frac{x(x^2-5x+3)}{2x}=\frac{x^2-5x+3}{2}\xrightarrow[x\to 0]{} \boxed{\tfrac{3}{2}}.
\]

\item $\displaystyle \lim_{x\to 1}\frac{x^2+7x-8}{x-1}$ \\
$ x^2+7x-8=(x-1)(x+8)$.
\[
\frac{(x-1)(x+8)}{x-1}=x+8\xrightarrow[x\to 1]{}\boxed{9}.
\]

\end{enumerate}

\end{frame}

\begin{frame}{\orange{Indeterminate forms of limits: the ratio of polynomials at infinite} (+0.2)}

    Given the real numbers $a_0, a_1, \dots, a_n$ and $b_0, b_1, \dots, b_m$ for $n, m\in\N$ and such that $a_n \neq 0$ and $b_m \neq 0$, define the polynomials
    \begin{align*}
        p(x) &= a_0 + a_1x + a_2x^2 + \dots + a_n x^n, \\ 
        q(x) &= b_0 + b_1x + b_2x^2 + \dots + b_m x^m.
    \end{align*}
    Proof that
    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]

        \item $\lim_{x\rightarrow \infty} \frac{p(x)}{q(x)} = 0$ if $n < m$
        
        \item $\lim_{x\rightarrow \infty} \frac{p(x)}{q(x)} = \infty$ if $n > m$
        
        \item $\lim_{x\rightarrow \infty} \frac{p(x)}{q(x)} = \frac{a_n}{b_m}$ if $n = m$
        
    \end{enumerate}
    
\end{frame}

\begin{frame}{Indeterminate forms of limits: the conjugation trick}

    Compute the limits:

    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]
        
        \item $\lim_{x\rightarrow 4}\frac{2 - \sqrt{x}}{4 - x}$

        \item $\lim_{x\rightarrow 0}\frac{\sqrt{x^2 - x + 4} - 2}{x^2 - 3x}$

        \item $\lim_{x\rightarrow 0}\frac{\sqrt{x^2 + 9x + 9} - 3}{x}$

        \item $\lim_{x\rightarrow 0}\frac{t}{\sqrt{4 - t} -2}$

    \end{enumerate}
    
\end{frame}

\begin{frame}{Indeterminate forms — conjugation trick: solutions}

\small
\begin{enumerate}[leftmargin=0.5cm, label=(\alph*), rightmargin=0.2cm]
\setlength\itemsep{0.3cm}
\item $\displaystyle \lim_{x\to 4}\frac{2-\sqrt{x}}{4-x}
=\frac{2-\sqrt{x}}{4-x}\cdot\frac{2+\sqrt{x}}{2+\sqrt{x}}
=\frac{4-x}{(4-x)(2+\sqrt{x})}
=\frac{1}{2+\sqrt{x}}\;\xrightarrow[x\to4]{}\;\boxed{\tfrac14}.$

\item $\displaystyle \lim_{x\to 0}\frac{\sqrt{x^2-x+4}-2}{x^2-3x}
=\frac{x^2-x}{(x^2-3x)(\sqrt{x^2-x+4}+2)}$ \\
\hspace{2.76cm} 
$=\frac{x(x-1)}{x(x-3)(\sqrt{x^2-x+4}+2)}
\;\xrightarrow[x\to0]{}\;\boxed{\tfrac{1}{12}}.$

\item $\displaystyle \lim_{x\to 0}\frac{\sqrt{x^2+9x+9}-3}{x}
=\frac{x^2+9x}{x(\sqrt{x^2+9x+9}+3)}
=\frac{x+9}{\sqrt{x^2+9x+9}+3}
\;\xrightarrow[x\to0]{}\;\boxed{\tfrac{3}{2}}.$

\item $\displaystyle \lim_{t\to 0}\frac{t}{\sqrt{4-t}-2}
=\frac{t(\sqrt{4-t}+2)}{(4-t)-4}
=-(\sqrt{4-t}+2)\;\xrightarrow[t\to0]{}\;\boxed{-4}.$

\end{enumerate}

\end{frame}

\begin{frame}{The sandwich technique (a.k.a squeeze theorem)}

    \begin{enumerate}[leftmargin = 0.5cm, label = \arabic*., rightmargin = 0.2cm]
        \item Suppose $y = f(x)$ is the equation of a curve that always lies between the parabola $x^2 = y - 1$ and the hyperbola $yx + y - 1 = 0$. What is the limit of $f(x)$ as $x\rightarrow 0$?

        \item Suppose $f(x)$ is a function lower than $g(x) = 1/2 + e^{-x^2}$ and greater than $h(x) = \frac{5x^2}{10(1+x^2) + 3x}$ for all $x \geq 0$. What is the limit of $f(x)$ as $x\rightarrow \infty$?
    \end{enumerate}
     
\end{frame}

\begin{frame}{The sandwich technique (a.k.a squeeze theorem) — Solution}

\small
\begin{enumerate}[leftmargin=0.5cm, label=\arabic*., rightmargin=0.2cm]

\item Parabola: $x^2=y-1 \Rightarrow y_1(x)=x^2+1$.
\quad Hyperbola: $yx+y-1=0 \Rightarrow y_2(x)=\dfrac{1}{1+x}$ (for $x\neq-1$).\\
“Between the curves” means $\min\{y_1(x),y_2(x)\}\le f(x)\le \max\{y_1(x),y_2(x)\}$.\\
$\displaystyle \lim_{x\to0}y_1(x)=1,\quad \lim_{x\to0}y_2(x)=1
\;\Rightarrow\; \boxed{\lim_{x\to0}f(x)=1}$. \\

\begin{center}
    \href{https://www.desmos.com/calculator/kwd4npf7tn}{Visualize here!}
\end{center}

\vspace{0.3cm}
\item For $x\ge 0$: $\displaystyle \frac{5x^2}{10(1+x^2)+3x}\ \le\ f(x)\ \le\ \frac12+e^{-x^2}$.\\
Right bound: $\displaystyle \lim_{x\to\infty}\Big(\tfrac12+e^{-x^2}\Big)=\tfrac12$.\\
Left bound:
\[
\lim_{x\to\infty}\frac{5x^2}{10(1+x^2)+3x}
=\lim_{x\to\infty}\frac{5}{10+\frac{3}{x}+\frac{10}{x^2}}=\tfrac12.
\]
Thus $\boxed{\lim_{x\to\infty}f(x)=\tfrac12}$. \\

\begin{center}
    \href{https://www.desmos.com/calculator/u9pxu613n6}{Visualize here!}
\end{center}

\end{enumerate}

\end{frame}


\begin{frame}{Intuition of continuity}

    Which of the following functions are likely to be continuous functions of time? What about their monotonicity?
    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]
        \item Price of an ounce of gold
        \item Height of a growing child
        \item Height of an airplane above sea level
        \item Distance traveled by a car
        \item Number of votes supporting a party counted during election night
    \end{enumerate}
    
\end{frame}

\begin{frame}{Intuition of continuity — Solutions}

\small
\begin{enumerate}[leftmargin=0.5cm, label=(\alph*), rightmargin=0.2cm]

\item \textbf{Price of gold:} \emph{Not continuous.}\\
Market prices change in discrete ticks; modeled as a piecewise constant function, which may increase or decrease at every jump.

\item \textbf{Height of a growing child:} \emph{Continuous.} \\
No instantaneous jumps in physical height; derivative (growth speed) may vary, but the function is continuous. It is increasing.

\item \textbf{Height of an airplane:} \emph{Continuous.}\\
Altitude cannot jump provided finite vertical speed; may increase or decrease.

\item \textbf{Distance traveled by a car:} \emph{Continuous and nondecreasing.}\\
It accumulates in a continuous fashion over time; it can only increase or stay constant.

\item \textbf{Votes counted share:} \emph{Not continuous.} \\
Increasing or constant, but updates arrive in batches (jumps).

\end{enumerate}

\end{frame}

\begin{frame}{Tangents}

    In each case, find the equation of the tangent at the specified point

    \begin{enumerate}[leftmargin = 0.5cm, label = (\alph*), rightmargin = 0.2cm]

        \item $f(x) = 3x + 2$, at $x = 0$

        \item $f(x) = x^2 - 1$, at $x= 1$

        \item $f(x) = 2 + 3/x^{-3}$, at $x = 3$

        \item $f(x) = \ln(x)$, at $x = e$

        \item $f(x) = e^{-x} + x$, at $x = 0$
        
    \end{enumerate}
    
\end{frame}

\begin{frame}{Tangents — Solutions}

\small
For $y=f(x)$ at $x=a$: \quad $\displaystyle y = f(a) + f'(a)\,(x-a)$.

\begin{enumerate}[leftmargin=0.5cm, label=(\alph*), rightmargin=0.2cm]

\item $f(x)=3x+2$, $a=0$.\quad $f'(x)=3$; $f(0)=2$.
\[
\boxed{\,y=3x+2\,}
\]

\item $f(x)=x^2-1$, $a=1$.\quad $f'(x)=2x$; $f'(1)=2$, $f(1)=0$.
\[
\boxed{\,y=0+2(x-1)=2x-2\,}
\]

\item $f(x)=2+\dfrac{3}{x^{-3}}=2+3x^{3}$, $a=3$.\quad $f'(x)=9x^{2}$; $f'(3)=81$, $f(3)=83$.
\[
\boxed{\,y=83+81(x-3)=81x-160\,}
\]

\item $f(x)=\ln x$, $a=e$.\quad $f'(x)=\dfrac{1}{x}$; $f'(e)=\dfrac{1}{e}$, $f(e)=1$.
\[
\boxed{\,y=1+\tfrac{1}{e}(x-e)=\tfrac{1}{e}x\,}
\]

\item $f(x)=e^{-x}+x$, $a=0$.\quad $f'(x)=-e^{-x}+1$; $f'(0)=0$, $f(0)=1$.
\[
\boxed{\,y=1+0\cdot(x-0)=1\,}
\]

\end{enumerate}
\end{frame}


\begin{frame}{Linear approximations}

    For $f(x) = x^n$, find an approximation of $f(x + 1)$ using the tangent line of $f$ at $x$, for $n\geq 1/2$. Proof that this approximation
    
    \begin{enumerate}[leftmargin=0.5cm, label=(\alph*), rightmargin=0.2cm]
    
        \item gets better as $x$ increases for $n = 1/2$; 
        
        \item is constant for $n = 2$;
        
        \item is zero when $n = 1$;

    \end{enumerate}

\end{frame}

\begin{frame}{Linear approximations — Solution}

\small

\textbf{Linear (first–order) approximation}. \\
Since $f'(x) = nx^{n-1}$ and $\Delta x = (x+1) - x = 1$,
\[
\boxed{\,f(x+1)\;\approx\; f(x) + f'(x)\Delta x = f(x)+n x^{n-1}
=\Big(1+\frac{n}{x}\Big)f(x)\,}.
\]

\textbf{Accuracy} \\
Error: \;\;$E_n(x)=f(x+1) - f(x)-n x^{n-1} = (x+1)^n - x^n - nx^{n-1}$.

\begin{enumerate}[leftmargin=0.5cm, label=(\alph*), rightmargin=0.2cm]

\setlength\itemsep{1em}

\item $n=\tfrac12$ (concave):\hfill \href{https://www.desmos.com/calculator/axrcmgyl4q}{Visualize here!}

    &= \sqrt{x+1} - \sqrt{x} - \frac{1}{2\sqrt{x}}

\end{align*}

    &= (x+1)^2 - x^2 - 2x 
    = x^2+2x+1 - x^2 - 2x
    = 1
\end{align*}

\item $n=1$ (linear): $E_1(x)=0$ and the approximation is exact

\end{enumerate}

\begin{frame}{The chain rule — Solution (1/2)}

\small
Assume differentiability where needed.

\begin{enumerate}[leftmargin=0.55cm, label=\arabic*., rightmargin=0.2cm]

\item Find a formula for $h'(x)$:
  \begin{enumerate}[leftmargin=0.5cm, label=(\alph*), rightmargin=0.2cm]
    \item $h(x)=f(x^2)$:
    \[
      \boxed{\,h'(x)=f'(x^2)\cdot 2x\,}
    \]
    \item $h(x)=f\!\big(x^n g(x)\big)$:

    \]

\begin{frame}{The chain rule — Solution (2/2)}

\small
\begin{enumerate}[leftmargin=0.55cm, label=\arabic*., start=3, rightmargin=0.2cm]

\item Suppose $U$ and $g$ are twice differentiable, increasing, and concave.  
Show $f(x)=g(U(x))$ is increasing and concave.

\textbf{Increasing:}
\[
f'(x)=g'(U(x))\,U'(x).
\]
Since $g'$ and $U'$ are $\ge 0$, we have $f'(x)\ge 0$ $\Rightarrow$ $f$ increasing.

\textbf{Concave:}
\[
f''(x)=g''(U(x))\,(U'(x))^2 + g'(U(x))\,U''(x).
\]
Concavity gives $g''\le 0$, $U''\le 0$; increasing gives $g'\ge 0$.  
Thus both terms are $\le 0$ $\Rightarrow$ $f''(x)\le 0$ $\Rightarrow$ $f$ concave.

\end{enumerate}

\end{frame}

\begin{frame}{Elasticity of demand}

    \begin{enumerate}[leftmargin = 0.5cm, label = \arabic*., rightmargin = 0.2cm]
        \item Determine the elasticity of demand, in terms of the price $p$, for the demand set
        $$
        D = \{(q, p)\ :\ q(1+p^2) = 100\}.
        $$
        For what values of $p$ is the demand inelastic?

        \item Proof that, for the demand function $q_D(p) = a - bp$, the demand is inelastic whenever $0 < p < a/(2b)$
    \end{enumerate}
    
\end{frame}

\begin{frame}{Elasticity of demand — Solution}
\footnotesize
\emph{Price elasticity of demand} at price $p$ is defined as
\[
\varepsilon(p)=q'(p)\cdot\frac{p}{q}\qquad
\text{(inelastic if }|\varepsilon(p)|<1\text{)}.
\]

\begin{enumerate}[leftmargin=0.5cm, label=\arabic*., rightmargin=0.2cm]
  \item \textbf{Demand: } $q(1+p^2)=100\ \Rightarrow\ q(p)=\dfrac{100}{1+p^2}$.
  \[
    q'(p)=-\frac{200p}{(1+p^2)^2},
    \qquad
    \varepsilon(p)=q'(p)\frac{p}{q}
      =\left(-\frac{200p}{(1+p^2)^2}\right)\!\left(\frac{p}{100/(1+p^2)}\right)
      =-\frac{2p^2}{1+p^2}.
  \]
  Hence $|\varepsilon(p)|=\dfrac{2p^2}{1+p^2}$, so demand is
  \[
    \text{inelastic when }|p|<1,\ \text{unit elastic at }|p|=1,\ \text{elastic when }|p|>1.
  \]
  Since $p\ge0$, this means for $0\le p<1$ the demand is inelastic.)

  \item \textbf{Linear demand: } $q_D(p)=a-bp$ with $a,b>0$.
  \[
    q'(p)=-b,\qquad
    \varepsilon(p)=q'(p)\frac{p}{q}
      =-b\,\frac{p}{a-bp},\qquad
    |\varepsilon(p)|=\frac{bp}{a-bp}.
  \]
  Then $|\varepsilon(p)|<1\iff bp<a-bp\iff 2bp<a\iff \boxed{\,0<p<\frac{a}{2b}\,}$
  %(also requiring $p<a/b$ so $q>0$).
\end{enumerate}
\end{frame}

\begin{frame}{Excise tax — Solution}
\footnotesize
Supply: $15p-2q=20 \ \Rightarrow\ q_S(p)=7.5\,p-10$. \quad
Demand: $q+5p=40 \ \Rightarrow\ q_D(p)=40-5p$.

\begin{enumerate}[leftmargin=0.5cm, label=(\alph*), rightmargin=0.2cm]
  \item \textbf{Equilibrium without tax.}
  \[
    7.5p-10=40-5p \ \Rightarrow\ 12.5p=50 \Rightarrow
    \boxed{p^*=4},\qquad \boxed{q^*=20}.
  \]

  \item \textbf{Per–unit tax $T$ on producers.}
  New supply: $q_{S_T}(p)=q_S(p-T)=7.5(p-T)-10=7.5p-7.5T-10$.
  \[
    7.5p-7.5T-10=40-5p \Rightarrow 12.5p=50+7.5T
    \Rightarrow \boxed{p^T=4+0.6T},\quad
    \boxed{q^T=40-5p^T=20-3T}.
  \]
  (Consumer price rises by $0.6$ per \$1 of tax; producers receive $p^T-T=4-0.4T$.)

  \item \textbf{Revenue maximization.}
  Government revenue: $R(T)=T\,q^T=T(20-3T)=20T-3T^2$ (concave).
  \[
    R'(T)=20-6T=0 \Rightarrow \boxed{T^{*}=\tfrac{10}{3}},\quad
    q^{T^*}=20-3\cdot\tfrac{10}{3}=10,\quad
    R(T^*)=\boxed{\tfrac{100}{3}}.
  \]
  Feasible taxes require $q^T\ge0 \Rightarrow T\le \tfrac{20}{3}$.
\end{enumerate}
\end{frame}

\begin{frame}{Optimal time to sell}

    You own a piece of land whose value $V(t)$ after $t$ years is $V(t)$ = $e^{\sqrt{t}}$. Assuming that interest on a bank deposit will be continuously compounded at the equivalent annual rate of $12.5\%$, write an expression for the present value of the amount realized by selling the land after $t$ years, and determine the optimum time to sell.
    
\end{frame}

\begin{frame}{Optimal time to sell — Solution}
% \footnotesize
Value at time $t$ (years): $V(t)=e^{\sqrt{t}}$. Continuous discount at rate $r=0.125$ per year gives the \emph{present value}
\[
\boxed{\;PV(t)=V(t)e^{-rt}=e^{\,\sqrt{t}-0.125\,t}\;},\qquad t\ge 0.
\]

To maximize, it suffices to maximize the exponent $\phi(t)=\sqrt{t}-0.125\,t$.
\[
\phi'(t)=\frac{1}{2\sqrt{t}}-0.125=0 \ \Longrightarrow\ \sqrt{t}=\frac{1}{2r}=\frac{1}{2\cdot0.125}=4
\ \Longrightarrow\ \boxed{t^*=16}.
\]
Since $\phi''(t)=-\dfrac{1}{4t^{3/2}}<0$, this is a (unique) maximum.  
Thus the optimal selling time is \(\boxed{t^*=16\text{ years}}\).

At $t^*$:
\[
PV(t^*)=e^{\,\sqrt{16}-0.125\cdot 16}=e^{\,4-2}=\boxed{e^{2}\approx 7.389}.%,\qquad
%V(16)=e^{4}\approx 54.598.
\]
