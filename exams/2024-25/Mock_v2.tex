% Question 1: Parametric Linear System
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\question[3]
Find the values of real parameters \(c\) and \(d\) for which the system has:
\[\begin{cases}
2x + y - z = 4,\\
x + y + z = 2,\\
x + cy + z = d,
\end{cases}\]
\begin{enumerate}[label=(\alph*)]
  \item No solution.
  \item Infinitely many solutions.
  \item Exactly one solution.
\end{enumerate}

\begin{solution}
Form the augmented matrix and row‑reduce:
\[
\left(\begin{array}{ccc|c}
2 & 1 & -1 & 4\\
1 & 1 &  1 & 2\\
1 & c &  1 & d
\end{array}\right)
\sim
\left(\begin{array}{ccc|c}
1 & 1 &  1 & 2\\
0 & -1& -3 & 0\\
0 & c-1 & 0 & d-2
\end{array}\right).
\]
The third row gives \((c-1)y = d-2\).
\begin{itemize}[nosep]
  \item If \(c=1\) and \(d-2\neq0\), equation impossible → no solution.
  \item If \(c=1\) and \(d=2\), third row is \(0=0\), two independent equations → infinitely many solutions.
  \item If \(c\neq1\), then \(y=(d-2)/(c-1)\) and back‑substitution yields a unique \((x,y,z)\).
\end{itemize}
\end{solution}

% Question 2: Derivative Approximation
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\question[2]
Let \(f(x)=x e^{-x}\).
\begin{enumerate}[label=(\alph*)]
  \item Compute \(f'(x)\).
  \item Use the derivative at \(x=0\) to give the linear approximation of \(f(0.05)\).
  \item Evaluate \(\lim_{x\to\infty}f(x)\).
\end{enumerate}

\begin{solution}
\begin{enumerate}[label=(\alph*)]
  \item \(f'(x)=e^{-x}(1 - x).\)
  \item At \(x=0\), \(f(0)=0\), \(f'(0)=1\).  So \(f(0.05)\approx f(0)+f'(0)\cdot0.05=0.05.\)
  \item As \(x\to\infty\), \(x e^{-x}\to0\) (exponential dominates polynomial).  Hence limit is 0.
\end{enumerate}
\end{solution}

% Question 3: Price Elasticity & Revenue
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\question[3]
A taxi service has daily demand \(D(p)=300-5p\), where \(p\) is the fare in euros and \(D\) the number of rides.
\begin{enumerate}[label=(\alph*)]
  \item Compute the price elasticity of demand \(E(p)=\frac{p}{D(p)}D'(p)\).
  \item Find the price \(p^*\) at which \(|E(p^*)|=1\) (unit elasticity).
  \item Give the formula of the daily revenue \(R(p)\) if all the daily demand is consumed.
  \item What value of $p$ maximizes the revenue?
\end{enumerate}

\begin{solution}
\begin{enumerate}[label=(\alph*)]
  \item \(D'(p)=-5\), so \(E(p)=\frac{p}{300-5p}(-5) = -\frac{5p}{300-5p}.\)
  \item If $300 - 5p > 0$, then 
  $$
  |E(p)|=1 \Leftrightarrow 5p=300-5p \Leftrightarrow 10p=300 \Leftrightarrow p^*=30.
  $$
  If $300 - 5p < 0$, then
  $$
  |E(p)|=1 \Leftrightarrow 5p=5p-300 \Leftrightarrow 0=300,
  $$
  meaning that there is no $p$ such that $300 - 5p < 0$ and $|E(p)|=1$ simultaneously. Hence, $p^* = 30$.
  \item \(R(p)=pD(p) = p(300-5p)=300p-5p^2\). 
  \item \(R'(p)=300-10p\), so \(R'(30)=0\) and \(R''(30)=-10<0\); hence \(p^*=30\) maximizes \(R\).
  This is aligned to the economic principle "unitary elasticity maximizes reward". 
\end{enumerate}
\end{solution}

% Question 4: Tank Filling Integrals
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\question[3]
A water tank is filled at an instantaneous rate
\[r(t)=0.5e^{2t-0.5t^2}\]
in cubic meters per hour, for \(0\le t\le24\).  Let \(V(t)\) be the volume function with \(V'(t)=r(t)\), such that 
\(V(0)=0\) and \(V(24)=9\).
\begin{enumerate}[label=(\alph*)]
  \item Compute the total volume of water added over 24 hours, \(\int_0^{24}r(t)\,dt\).
  \item Show that \(r'(t)=(1-\tfrac t2)e^{2t-0.5t^2}\), and find the time \(t^*\) when the filling rate is maximized.
  \item If the filling process stops at \(t=12\) hours instead, does the maximum rate occur at a different time? Explain.
\end{enumerate}

\begin{solution}
\begin{enumerate}[label=(\alph*)]
  \item By the Fundamental Theorem of Calculus,
  \[
    \int_0^{24}r(t)\,dt = V(24)-V(0) = 9.0\ \text{m}^3.
  \]
  \item Differentiate:
  \(
    r'(t)=0.5e^{2t-0.5t^2}(2 - t)=e^{2t-0.5t^2}(1 - \tfrac t2).
  \)
  Set \(1-\tfrac t2=0\) → \(t=2\) hours.  Since \(r'(t)>0\) for \(t<2\) and \(r'(t)<0\) for \(t>2\), this is the global maximum.
  \item Restricting \(t\in[0,12]\), the critical point \(t^*=2\) still lies in the interval, so the time of maximum remains \(2\) hours.
\end{enumerate}
\end{solution}
