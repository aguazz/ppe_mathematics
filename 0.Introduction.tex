    \begin{block}{Module 3: Integral calculus}
        \begin{minipage}{0.35\textwidth}
            \centering
            \includegraphics[width = 0.9\textwidth]{img/integral.png}    
        \end{minipage}
        \begin{minipage}{0.6\textwidth}
            \begin{itemize}[leftmargin = 0.5cm, label = $\bullet$]
